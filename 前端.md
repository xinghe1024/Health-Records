### 运行配置

```text
"scripts": {
    "dev": "concurrently -k -n API,WEB -c yellow,cyan \"npm:server\" \"npm:client\"",
    "client": "vite --host 0.0.0.0",
    "server": "node server/server.js",
    "build": "vite build",
    "preview": "vite preview --host 0.0.0.0"
}
-k: 参数，表示kill other processes when one exits
-n API,WEB: 给进程命名，用逗号分隔，这里有两个名字API和WEB
-c yellow,cyan: 设置颜色，对应命名进程的输出颜色，yellow给API，cyan给WEB。
```

### 状态管理

```text
import { defineStore } from 'pinia'
import { computed, ref } from 'vue'
import { healthApi } from '../api'

export const useHealthStore = defineStore('health', () => {
  const patientId = ref(1)
  const patients = ref([])
  const dashboard = ref(null)
  const loading = ref(false)
  const currentPatient = computed(() => patients.value.find(p => p.id === patientId.value) || dashboard.value?.patient)

  async function init() {
    loading.value = true
    try {
      patients.value = await healthApi.patients()
      dashboard.value = await healthApi.dashboard(patientId.value)
    } finally { loading.value = false }
  }
  async function changePatient(id) {
    patientId.value = id
    dashboard.value = await healthApi.dashboard(id)
  }
  async function refreshDashboard() { dashboard.value = await healthApi.dashboard(patientId.value) }

  return { patientId, patients, dashboard, currentPatient, loading, init, changePatient, refreshDashboard }
})
```
